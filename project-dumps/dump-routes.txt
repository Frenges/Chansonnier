<script lang="ts">
  import { onMount } from 'svelte';
  import { writable } from 'svelte/store';

  // Mode nuit stocké dans dataset root
  const theme = writable('light');
  export const offlineReady = writable(false);

  onMount(async () => {
    // Restaurer thème depuis localStorage
    const saved = localStorage.getItem('theme');
    if (saved) {
      theme.set(saved);
      document.documentElement.dataset.theme = saved;
    }

    // Enregistrer le service worker uniquement en production
    if ('serviceWorker' in navigator && import.meta.env.PROD) {
      try {
        await navigator.serviceWorker.register('/service-worker.js');
        console.log('SW registered');
      } catch (e) {
        console.warn('SW registration failed', e);
      }
    }

    // Populate IndexedDB uniquement côté client et de façon non bloquante
    // Import dynamique pour éviter tout import client-side en SSR
    try {
      const mod = await import('$lib/dexie-client');
      mod.ensureDexieIsPopulated().then(() => offlineReady.set(true)).catch(err => console.warn('Dexie population failed', err));
    } catch (e) {
      // si le module n'existe pas, on ignore
    }
  });

  // Theme toggle helper
  function toggleTheme() {
    const current = document.documentElement.dataset.theme === 'dark' ? 'light' : 'dark';
    document.documentElement.dataset.theme = current;
    localStorage.setItem('theme', current);
    theme.set(current);
  }
</script>
