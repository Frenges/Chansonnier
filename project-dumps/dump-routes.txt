<script lang="ts">
  // Le layout reçoit data depuis +layout.server.ts
  export let data: { pages?: any[] };

  import { onMount } from 'svelte';
  import { writable, derived } from 'svelte/store';
  import { goto } from '$app/navigation';
  import "../app.css";

  // Theme
  const theme = writable('light');

  // Recherche
  const query = writable('');
  const pagesStore = writable(data?.pages ?? []);

  // Met à jour pagesStore si data change (SSR -> client hydration)
  $: if (data?.pages) pagesStore.set(data.pages);

  const results = derived([pagesStore, query], ([$pages, $query]) => {
    if (!$pages) return [];
    if (!$query || $query.trim() === '') return $pages.slice(0, 50);
    const q = $query.toLowerCase();
    return $pages.filter(p =>
      ((p.title || '').toLowerCase().includes(q)) ||
      ((p.slug || '').toLowerCase().includes(q))
    ).slice(0, 50);
  });

  // SW / Dexie / offline : import dynamique côté client
  export const offlineReady = writable(false);

  onMount(async () => {
    // restore theme
    const saved = localStorage.getItem('theme');
    if (saved) {
      theme.set(saved);
      document.documentElement.dataset.theme = saved;
    }

    // register SW only in production
    if ('serviceWorker' in navigator && import.meta.env.PROD) {
      try {
        await navigator.serviceWorker.register('/service-worker.js');
        console.log('SW registered');
      } catch (e) {
        console.warn('SW registration failed', e);
      }
    }

    // import dexie-client dynamically (non-blocking)
    try {
      const mod = await import('$lib/dexie-client');
      mod.ensureDexieIsPopulated().then(() => offlineReady.set(true)).catch(err => console.warn('Dexie population failed', err));
    } catch (e) {
      // ignore in dev if module missing
    }
  });

  function toggleTheme() {
    const current = document.documentElement.dataset.theme === 'dark' ? 'light' : 'dark';
    document.documentElement.dataset.theme = current;
    localStorage.setItem('theme', current);
    theme.set(current);
  }

  // navigation helper (use goto to avoid full reload)
  function openResult(r: any) {
    const slug = r.slug ?? r.id;
    if (!slug) return;
    goto(`/page/${slug}`);
  }
</script>
